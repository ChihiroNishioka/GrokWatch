cmake_minimum_required(VERSION 3.16)
project(grokwatch_host LANGUAGES CXX)
enable_testing()
option(GROKWATCH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(GROKWATCH_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(core_dir "${CMAKE_CURRENT_SOURCE_DIR}/../components/grokwatch_core")
if(EXISTS "${core_dir}/CMakeLists.txt")
    add_subdirectory("${core_dir}" core)
endif()
file(GLOB suites CONFIGURE_DEPENDS test_*.cpp)
foreach(source IN LISTS suites)
    get_filename_component(suite "${source}" NAME_WE)
    add_executable(${suite} "${source}")
    target_compile_features(${suite} PRIVATE cxx_std_17)
    target_compile_options(${suite} PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
    target_include_directories(${suite} PRIVATE "${core_dir}/include")
    if(TARGET grokwatch_core)
        target_link_libraries(${suite} PRIVATE grokwatch_core)
    endif()
    file(STRINGS "${source}" cases REGEX "^TEST\\(")
    foreach(test_case IN LISTS cases)
        string(REGEX REPLACE "^TEST\\(([a-z0-9_]+)\\).*" "\\1" name "${test_case}")
        add_test(NAME "${name}" COMMAND ${suite} "${name}")
    endforeach()
endforeach()
add_test(NAME check_core_includes COMMAND ${CMAKE_COMMAND} "-DCORE=${core_dir}"
         -P "${CMAKE_CURRENT_SOURCE_DIR}/check_core_includes.cmake")
